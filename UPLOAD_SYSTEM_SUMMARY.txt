تم إنشاء أجوبة شاملة على أسئلتك الثلاثة:

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1️⃣  هل مكتبة Multer مثبتة؟
   ────────────────────────────────────
   ❌ لا - وليست مطلوبة
   
   السبب:
   • المشروع يستخدم Base64 Encoding (أكثر أماناً)
   • يستخدم TRPC بدلاً من Express Multer
   • Firebase Storage مدمج بالفعل
   
   التثبيت الحالي:
   ✅ express: ^4.21.2
   ✅ firebase-admin: ^14.0.0
   ✅ TRPC: ^11.6.0
   ❌ multer: غير موجود (غير ضروري)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

2️⃣  كود السيرفر الكامل (استقبال الرفع)
   ────────────────────────────────────
   📁 الموقع: server/routers/uploads.ts
   
   المسؤوليات:
   ✓ استقبال البيانات المشفرة بـ Base64
   ✓ تحويل Base64 إلى Buffer
   ✓ التحقق من حجم الملف (max 10 MB)
   ✓ إنشاء مسار فريد (unique path)
   ✓ رفع للـ Firebase Storage
   ✓ fallback للتخزين المحلي في حالة الفشل
   ✓ حفظ البيانات في قاعدة البيانات
   
   العمليات الأساسية:
   1. Base64 → Buffer: Buffer.from(input.fileData, "base64")
   2. التحقق: if (fileSize > 10 * 1024 * 1024)
   3. المسار الفريد: `user-${userId}/purpose/${randomId}.ext`
   4. التخزين: storagePut(fileKey, buffer, mimeType)
   5. قاعدة البيانات: createFileUpload({...})

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

3️⃣  كود الإرسال من الواجهة الأمامية
   ────────────────────────────────────
   
   الطريقة الموصى بها (استخدام المكون):
   ─────────────────────────────────────
   
   import { FileUploadComponent } from '@/components/FileUploadComponent';
   
   <FileUploadComponent
     purpose="profile-picture"
     maxSizeMB={5}
     acceptedFileTypes="image/*"
     onUploadSuccess={(url, filename) => {
       console.log('تم الرفع:', url);
     }}
   />
   
   ✓ واجهة رسومية كاملة
   ✓ معالجة الأخطاء التلقائية
   ✓ Drag & Drop مدعوم
   ✓ شريط التقدم (Loading state)
   
   ─────────────────────────────────────
   الطريقة اليدوية (إذا لزم الأمر):
   ─────────────────────────────────────
   
   // 1. تحويل الملف إلى Base64
   const arrayBuffer = await file.arrayBuffer();
   const base64 = btoa(String.fromCharCode(...new Uint8Array(arrayBuffer)));
   
   // 2. إرسال للسيرفر
   const response = await fetch('/api/trpc/uploads.upload', {
     method: 'POST',
     headers: { 'Content-Type': 'application/json' },
     body: JSON.stringify({
       fileData: base64,
       filename: file.name,
       mimeType: file.type,
       purpose: 'general'
     })
   });
   
   // 3. الحصول على الرابط
   const { result } = await response.json();
   console.log('رابط الملف:', result.url);
